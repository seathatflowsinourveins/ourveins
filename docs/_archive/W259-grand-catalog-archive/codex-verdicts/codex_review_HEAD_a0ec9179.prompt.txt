+#!/usr/bin/env python3
+# FM-20 path-drift cascade — PreToolUse:Agent lint (advisory; DRY-RUN)
+#
+# Design source: W152-F13 codex T1 fm20_design block at
+#   .claude/state/codex_consult_wave152_f13_fm_lints_design_OUT.txt L10524
+#   (APPROVE conf=0.84; LOC=80 estimate; feasibility=med; DRY-RUN/warn for first rollout)
+# Discipline source: Z:/claude-sota/.claude/rules/fm20-path-drift-cascade.md (sister cite-import-AMBER)
+# Schema: PreToolUse hook input per https://code.claude.com/docs/en/hooks
+"""Detect TIER-3 claim propagation in Agent dispatch briefs without verification markers.
+
+Scope:
+  - PreToolUse:Agent only — inspects Agent tool_input.prompt + .description
+  - DRY-RUN: emits JSONL + stderr warn; does NOT block dispatch
+  - Future: tighten to decision:block once marker convention proves consistent
+
+Detection (TIER-3 propagation patterns):
+  - Agent verdict text (NEEDS-REVISION/APPROVE/REJECT with conf=0.NN)
+  - tmp/wave*.md paths
+  - codex_consult*_OUT.txt paths
+  - memory/feedback_*.md paths
+  - task-tracking claims (task #\\d+ / [completed] / [pending])
+  - "Agent X said/found/recommended/caught/surfaced/flagged"
+
+Verification proximity check:
+  - For each TIER-3 detection: search ±200 chars for [VERIFIED]/[MEASURED]/Mia probe:/REFUTED/DROPPED
+  - Marker nearby → CLASSIFY-OK (no warn)
+  - Marker missing → CLASSIFY-FM-20-RISK (emit JSONL + stderr warn; exit 0)
+
+Per fm20_design "fp_risk: med": DRY-RUN-WARN mode for first rollout.
+"""
+from __future__ import annotations
+
+import os
+import re
+import sys
+
+sys.path.insert(0, os.path.dirname(__file__))
+from _observation_writer import (  # noqa: E402
+    append_observation,
+    now_iso,
+    read_stdin_payload,
+)
+
+_CTX = "fm20_path_drift_lint"
+_DISABLE_ENV = "FM20_PATH_DRIFT_LINT_DISABLE"
+_NEARBY_WINDOW = 200  # chars on each side to search for verification marker
+_MAX_FINDINGS_LOGGED = 10
+
+# TIER-3 propagation patterns (cls, compiled regex)
+_PROPAGATION = [
+    ("verdict_conf", re.compile(r"(?:NEEDS-REVISION|NEEDS-ATTENTION|APPROVE|REJECT)\b[^.\n]{0,40}?\bconf\s*=\s*0\.\d{2}", re.IGNORECASE)),
+    ("tmp_wave",     re.compile(r"tmp/wave[\w\-./]+\.md", re.IGNORECASE)),
+    ("codex_out",    re.compile(r"codex_consult[\w\-./]*_OUT\.txt", re.IGNORECASE)),
+    ("feedback_mem", re.compile(r"memory/feedback_[\w\-./]+\.md", re.IGNORECASE)),
+    ("task_claim",   re.compile(r"task\s*#\d+|\[completed\]|\[pending\]")),
+    ("agent_said",   re.compile(r"Agent\s+[A-Z]\w*\s+(?:found|said|recommended|caught|surfaced|flagged|reported|claimed)", re.IGNORECASE)),
+]
+
+# Verification marker (any one nearby satisfies the proximity check)
+_VERIFICATION = re.compile(
+    r"\[VERIFIED\b|\[MEASURED\b|Mia probe:|REFUTED|DROPPED",
+    re.IGNORECASE,
+)
+
+
+def _has_nearby_marker(text: str, start: int, end: int) -> bool:
+    lo = max(0, start - _NEARBY_WINDOW)
+    hi = min(len(text), end + _NEARBY_WINDOW)
+    return bool(_VERIFICATION.search(text[lo:hi]))
+
+
+def main() -> int:
+    payload = read_stdin_payload(_DISABLE_ENV, _CTX)
+    if payload is None:
+        return 0
+
+    tool_input = payload.get("tool_input") or {}
+    if not isinstance(tool_input, dict):
+        return 0
+
+    prompt = str(tool_input.get("prompt") or "")
+    description = str(tool_input.get("description") or "")
+    text = description + "\n\n" + prompt
+    if not text.strip():
+        return 0
+
+    findings = []
+    for cls, pat in _PROPAGATION:
+        for m in pat.finditer(text):
+            if not _has_nearby_marker(text, m.start(), m.end()):
+                findings.append({"class": cls, "match": m.group(0)[:80], "pos": m.start()})
+
+    if not findings:
+        return 0  # Clean — no FM-20 risk detected
+
+    obs = {
+        "ts": now_iso(),
+        "ctx": _CTX,
+        "subagent_type": str(tool_input.get("subagent_type") or ""),
+        "findings_count": len(findings),
+        "findings": findings[:_MAX_FINDINGS_LOGGED],
+        "mode": "DRY-RUN-WARN",
+    }
+    try:
+        append_observation(obs, _CTX)
+    except Exception:
+        pass
+
+    sys.stderr.write(
+        f"[fm20-lint] FM-20 risk: {len(findings)} TIER-3 propagation(s) in Agent brief "
+        f"WITHOUT nearby [VERIFIED/MEASURED/Mia probe/REFUTED] marker. "
+        f"Consider decomposing sub-claims + Mia probe before dispatch. (DRY-RUN; not blocking)\n"
+    )
+    return 0
+
+
+if __name__ == "__main__":
+    sys.exit(main())

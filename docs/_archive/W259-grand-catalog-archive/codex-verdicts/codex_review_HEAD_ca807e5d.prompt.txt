+#!/usr/bin/env python3
+# FM-17 a-f sub-class classifier — SubagentStop hook (advisory; exit 0)
+#
+# Design source: W152-F13 codex T1 fm17_design block at
+#   .claude/state/codex_consult_wave152_f13_fm_lints_design_OUT.txt EOF
+#   (APPROVE conf=0.84; ~95 LOC; HIGH feasibility)
+# Sub-class catalog: .claude/rules/fm17-subagent-fleet-depletion.md
+# Schema: SubagentStop input per claude-agent-sdk-python types.py:309-316
+"""Classify subagent termination into FM-17 sub-classes (a-f).
+
+Precedence (most specific first):
+  f    1M-context billing                "Extra usage is required for 1M context"
+  b    429 pool depletion                "429" / "cooling down"
+  e    Autocompact thrashing             "Autocompact is thrashing"
+  c.ii Windows cert-store ACL            "certificate store" / "Access is denied"
+  c.i  Codex bg-job wedge                "task forwarded to Codex companion"
+  d    Wrapper stream-watchdog stall     "stream watchdog" / "Agent stalled" / INTERNAL_ERROR
+  a    Wrapper truncation                last_msg ends mid-sentence (heuristic fallback)
+
+Advisory-only: emits stderr + JSONL row when a sub-class matches; never blocks.
+Quiet exit if no signature found (no telemetry noise).
+"""
+from __future__ import annotations
+
+import os
+import re
+import sys
+from pathlib import Path
+
+sys.path.insert(0, os.path.dirname(__file__))
+from _observation_writer import (  # noqa: E402
+    append_observation,
+    now_iso,
+    read_stdin_payload,
+)
+
+_CTX = "fm17_class_lint"
+_DISABLE_ENV = "FM17_CLASS_LINT_DISABLE"
+_TAIL_BYTES = int(os.getenv("FM17_TAIL_BYTES", "262144"))
+
+# Sub-class signatures (precedence order — most specific first)
+_SIGNATURES = [
+    ("f",    r"Extra usage is required for 1M context"),
+    ("b",    r"\b429\b|cooling down"),
+    ("e",    r"Autocompact is thrashing"),
+    ("c.ii", r"certificate store|Access is denied|failed to open current user certificate store"),
+    ("c.i",  r"task forwarded to Codex companion"),
+    ("d",    r"stream watchdog did not recover|Agent stalled: no progress for|INTERNAL_ERROR"),
+]
+
+_RECOVERY = {
+    "a":    "SendMessage agent for ARTIFACT-INLINE re-emit",
+    "b":    "Fleet probe >=3 accounts <50% via status.py before re-dispatch",
+    "c.i":  "Foreground+tee from main session bypasses bg-job state wedge",
+    "c.ii": "Dispatch from main orchestrator process (cert-store ACL inheritance)",
+    "d":    "Foreground+tee from main OR re-brief with 90-180s per-codex-call budget",
+    "e":    "Tighten brief: substitute Read/WebFetch with ctx_execute_file/ctx_fetch_and_index",
+    "f":    "Path P codex exec foreground+tee OR Path D CLAUDE_CODE_DISABLE_1M_CONTEXT=1",
+}
+
+
+def _read_tail(path) -> str:
+    if not path:
+        return ""
+    try:
+        p = Path(path)
+        if not p.exists() or not p.is_file():
+            return ""
+        size = p.stat().st_size
+        with p.open("rb") as f:
+            if size > _TAIL_BYTES:
+                f.seek(size - _TAIL_BYTES)
+                f.readline()
+            return f.read().decode("utf-8", "replace")
+    except Exception:
+        return ""
+
+
+def _classify(text: str):
+    if not text:
+        return None, None
+    for sub, pat in _SIGNATURES:
+        m = re.search(pat, text, re.IGNORECASE)
+        if m:
+            return sub, m.group(0)[:80]
+    return None, None
+
+
+def _detect_truncation(last_msg: str) -> bool:
+    """Heuristic class 'a': last_msg ends mid-sentence without proper closure."""
+    if not last_msg or len(last_msg) < 200:
+        return False
+    tail = last_msg.rstrip()[-80:]
+    if re.search(r'[.!?\]\)`"\'>]$|---$|\*\*\*$|</\w+>$', tail):
+        return False
+    return True
+
+
+def main() -> int:
+    payload = read_stdin_payload(_DISABLE_ENV, _CTX)
+    if payload is None:
+        return 0
+
+    agent_id = str(payload.get("agent_id") or "")
+    agent_type = str(payload.get("agent_type") or "")
+    transcript_path = payload.get("agent_transcript_path")
+    last_msg = str(payload.get("last_assistant_message") or "")
+
+    transcript_tail = _read_tail(transcript_path)
+    sub, sig = _classify(last_msg)
+    if sub is None:
+        sub, sig = _classify(transcript_tail)
+    if sub is None and _detect_truncation(last_msg):
+        sub, sig = "a", "[heuristic: last_msg ends mid-sentence]"
+
+    if sub is None:
+        return 0  # Quiet exit — no FM-17 signature
+
+    obs = {
+        "ts": now_iso(),
+        "ctx": _CTX,
+        "agent_id": agent_id,
+        "agent_type": agent_type,
+        "fm17_subclass": sub,
+        "signature": sig,
+        "recovery_hint": _RECOVERY.get(sub, ""),
+    }
+    try:
+        append_observation(obs, _CTX)
+    except Exception:
+        pass
+
+    sys.stderr.write(
+        f"[fm17-lint] FM-17.{sub} detected for {agent_type or 'agent'} "
+        f"(sig: {(sig or '')[:60]}). Recovery: {_RECOVERY.get(sub, 'see fm17 rule')}\n"
+    )
+    return 0
+
+
+if __name__ == "__main__":
+    sys.exit(main())

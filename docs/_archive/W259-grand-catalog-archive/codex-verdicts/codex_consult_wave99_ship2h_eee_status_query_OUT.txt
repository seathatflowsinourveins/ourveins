def parse_dt(s):
    if not s:
        return None
    s = s.replace('Z', '+00:00')
    try:
        return datetime.fromisoformat(s)
    except ValueError:
        try:
            return datetime.strptime(s.split('.')[0], '%Y-%m-%d %H:%M:%S').replace(tzinfo=timezone.utc)
        except ValueError:
            return None


def fmt_delta(td):
    if td is None:
        return '-'
    s = int(td.total_seconds())
    if s < 0:
        return f'EXPIRED {-s // 3600}h ago'
    if s < 3600:
        return f'{s // 60}m'
    if s < 86400:
        return f'{s // 3600}h{(s % 3600) // 60}m'
    return f'{s // 86400}d{(s % 86400) // 3600}h'

def probe_cpa_api():
    req = urllib.request.Request(
        f'{CPA_BASE}/v0/management/auth-files',
        headers={'Authorization': f'Bearer {CPA_AUTH}'},
    )
    try:
        with urllib.request.urlopen(req, timeout=CPA_TIMEOUT) as resp:
            body = resp.read().decode('utf-8', errors='replace')
            try:
                return {'status': 'ok', 'data': json.loads(body)}
            except json.JSONDecodeError:
                return {'status': 'parse_error', 'raw': body[:500]}
    except urllib.error.HTTPError as e:
        return {'status': 'http_error', 'code': e.code, 'reason': str(e.reason)}
    except (urllib.error.URLError, TimeoutError, ConnectionError, OSError) as e:
        return {'status': 'down', 'error': str(e)}


def render_table(data):
    if data.get('error'):
        return f'ERROR: {data["error"]}'
    rows = data.get('rows', [])
    if not rows:
        return '(no rows)'
    now = datetime.now(timezone.utc)
    out = []
    hdr = f'{"name":35s} {"type":12s} {"plan":6s} {"5h%":4s} {"5h-reset":12s} {"7d%":4s} {"7d-reset":12s} {"req":4s} {"ok":4s} {"fail":5s} {"cache_tok":12s} {"total_tok":10s} {"status":7s}'
    out.append(hdr)
    out.append('-' * len(hdr))
    for r in rows:
        p_reset = parse_dt(r.get('primary_window_reset_at'))
        s_reset = parse_dt(r.get('secondary_window_reset_at'))
        p_in = fmt_delta(p_reset - now) if p_reset else '-'
        s_in = fmt_delta(s_reset - now) if s_reset else '-'
        p_pct = r.get('primary_window_used_percent')
        s_pct = r.get('secondary_window_used_percent')
        p_str = f'{p_pct}%' if p_pct is not None else '-'
        s_str = f'{s_pct}%' if s_pct is not None else '-'
        plan = (r.get('plan_type') or '-')[:6]
        status = 'LIMIT' if r.get('limit_reached') else 'ok'
        if (s_pct is not None and s_pct >= 90) or status == 'LIMIT':
            status_disp = '🔴 LIM' if status == 'LIMIT' else '🟡 hi'
        elif s_pct is not None and s_pct >= 70:
            status_disp = '🟡 hi'
        else:
            status_disp = '✅ ok'
        out.append(
            f'{r["name"][:34]:35s} {r["type"]:12s} {plan:6s} {p_str:4s} {p_in:12s} {s_str:4s} {s_in:12s} '
            f'{r["total_requests"]:4d} {r["success_count"]:4d} {r["failure_count"]:5d} '
            f'{r["cached_tokens"]:12d} {r["total_tokens"]:10d} {status_disp:7s}'
        )
    return '\n'.join(out)


def render_cpa_status(probe):
    s = probe.get('status', 'unknown')
    if s == 'ok':
        n = len(probe.get('data', {}).get('auth_files', [])) if isinstance(probe.get('data'), dict) else '?'
        return f'CPA: UP — /v0/management/auth-files returned {n} entries'
    if s == 'down':
        return f'CPA: DOWN at {CPA_BASE} — {probe.get("error", "unknown")[:80]}'
    return f'CPA: {s} — {probe.get("error", probe.get("reason", ""))[:80]}'

def update_cpa_sqlite(db_path: Path, results: list[PollResult], dry_run: bool = False) -> dict:
    if not db_path.exists():
        return {"error": "db_missing", "path": str(db_path)}
    stats = {"updated": 0, "no_match": 0, "skipped": 0, "errors": []}
    if dry_run:
        for r in results:
            if r.status == "ok":
                stats["updated"] += 1
            else:
                stats["skipped"] += 1
        stats["dry_run"] = True
        return stats
    conn = sqlite3.connect(str(db_path))
    try:
        for r in results:
            if r.status != "ok":
                stats["skipped"] += 1
                continue
            email_key = r.account.split(" (")[0]
            try:
                # F-3: capture cursor.rowcount to distinguish actual UPDATE
                # from no-row-matched (e.g., name/provider mismatch)
                cur = conn.execute(
                    """
                    UPDATE usage_identities SET
                        primary_window_used_percent = ?,
                        primary_window_limit_seconds = ?,
                        primary_window_reset_at = ?,
                        secondary_window_used_percent = ?,
                        secondary_window_limit_seconds = ?,
                        secondary_window_reset_at = ?,
                        plan_type = ?,
                        limit_reached = ?,
                        stats_updated_at = ?
                    WHERE name = ? AND provider = ?
                    """,
                    (
                        int(r.primary_used_percent) if r.primary_used_percent is not None else None,
                        r.primary_limit_seconds,
                        r.primary_reset_at,
                        int(r.secondary_used_percent) if r.secondary_used_percent is not None else None,
                        r.secondary_limit_seconds,
                        r.secondary_reset_at,
                        r.plan_type,
                        1 if r.limit_reached else 0 if r.limit_reached is not None else None,
                        datetime.now(timezone.utc).isoformat(),
                        email_key,
                        r.provider,
                    ),
                )
                if cur.rowcount >= 1:
                    stats["updated"] += 1
                else:
                    stats["no_match"] += 1
                    stats["errors"].append(
                        {"account": r.account, "provider": r.provider, "error": "no_row_matched_name_provider"}
                    )
            except Exception as e:
                stats["errors"].append({"account": r.account, "error": str(e)[:80]})
        conn.commit()
    finally:
        conn.close()
    return stats

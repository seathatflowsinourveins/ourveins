def _strip_quoted_regions(cmd: str) -> str:
    """Replace quoted-string contents with a placeholder so downstream pattern
    matchers see shell structure without matching inside quoted narrative data.

    Cycle-187 verified: real flags kept; narrative stripped to placeholder.
    Closes cycle-186 documented known-limitation (commit message containing
    literal destructive syntax).
    """
    return re.sub(r"'[^']*'", "'X'", re.sub(r'"[^"]*"', '"X"', cmd))


# Commands whose args are DATA (quoted text), not executable subcommands.
# Skip pattern-match to avoid false-positives on commit messages / echoed text.
_DATA_PREFIXES: tuple[re.Pattern[str], ...] = (
    re.compile(r"^\s*git\s+commit\b"),
    re.compile(r"^\s*echo\s+"),
    re.compile(r"^\s*printf\s+"),
    re.compile(r"^\s*gh\s+(issue|pr|release|api)\s+(create|comment|edit|review|close|merge)\b"),
    re.compile(r"^\s*cat\s*>\s*\S+\s*<<"),  # heredoc write
)


# Split on shell command separators — but NOT on newlines, because newlines inside
# quoted heredocs / commit messages are legitimate text.
_SPLIT_RE = re.compile(r"(?:;|&&|\|\|)+")

# Always-block patterns that fire even inside _DATA_PREFIX exempt commands.
_ALWAYS_BLOCK: tuple[tuple[re.Pattern[str], str, str], ...] = (
    (re.compile(r"--no-verify\b"),
     "--no-verify (skips pre-commit/pre-push hooks)",
     "fix the underlying hook failure; never skip safety hooks"),
)

# Cycle-186: destructive patterns the per-sub splitter cannot catch because the pattern

# scan; full _DESTRUCTIVE list is correctly handled by per-sub logic.
_WHOLE_COMMAND_ONLY_DESTRUCTIVE: tuple[tuple[re.Pattern[str], str, str], ...] = (
    (re.compile(r":\(\)\s*\{\s*:\s*\|\s*:\s*&\s*\}\s*;\s*:"),
     "fork bomb",

)


# Text-handling tools whose args are PURE DATA — exempt from _ALWAYS_BLOCK so
# `grep -E "--no-verify"` or `echo "--no-verify"` in narrative text are ALLOWED.
_ALWAYS_BLOCK_EXEMPT_PREFIXES: tuple[re.Pattern[str], ...] = (
    re.compile(r"^\s*echo\s+"),
    re.compile(r"^\s*printf\s+"),
    re.compile(r"^\s*grep\b"),
    re.compile(r"^\s*rg\b"),
    re.compile(r"^\s*cat\s+"),
    re.compile(r"^\s*awk\b"),
    re.compile(r"^\s*sed\b"),
    re.compile(r"^\s*less\b"),
    re.compile(r"^\s*more\b"),
)


def _iter_subcommands(command: str):
    """Split a multi-line shell command into individual executable sub-commands.

    Skips comment lines (leading `#`) and blank lines. Splits on `|` (pipe) so
    destructive in tail of pipe is caught.
    """
    cleaned_lines = []
    for line in command.splitlines():
        s = line.strip()
        if s and not s.startswith("#"):
            cleaned_lines.append(line)
    cleaned = "\n".join(cleaned_lines) if cleaned_lines else command
    for part in _SPLIT_RE.split(cleaned):
        stripped = part.strip()
        if not stripped:
            continue
        for piped in stripped.split("|"):
            piped = piped.strip()
            if piped:
                yield piped

    for cand in GITLEAKS_BIN_CANDIDATES:

    return None


def _resolve_repo_root() -> Path | None:
    """Find git repo root via CLAUDE_PROJECT_DIR or git rev-parse."""
    cpd = os.environ.get("CLAUDE_PROJECT_DIR")
    if cpd and (Path(cpd) / ".git").exists():
        return Path(cpd)
    try:
        result = subprocess.run(
            ["git", "rev-parse", "--show-toplevel"],
            capture_output=True, text=True, timeout=3,
        )
        if result.returncode == 0:
            return Path(result.stdout.strip())
    except Exception:
        pass
    return None


def _read_hook_input() -> dict:
    """Parse JSON stdin per CC PreToolUse hook contract."""
    try:

    except Exception:

def _is_git_commit_command(tool_input: dict) -> bool:
    """Match `git commit *` AND `git -C <path> commit *` (per codex T1 F3)."""
    cmd = tool_input.get("command", "")
    if not isinstance(cmd, str):
        return False
    # Fast-path: starts with `git commit` (most common)
    stripped = cmd.lstrip()
    if stripped.startswith("git commit"):
        return True
    # `git -C <path> commit ...` form
    if stripped.startswith("git -C"):
        # crude tokenize: "git -C <path> commit ..."
        toks = stripped.split(None, 4)
        if len(toks) >= 4 and toks[3] == "commit":
            return True
    return False

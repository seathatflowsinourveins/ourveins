def http_get_json(url: str, headers: dict, timeout: int = 15) -> tuple[int, dict | str]:
    req = urllib.request.Request(url, headers=headers)
    try:
        resp = urllib.request.urlopen(req, timeout=timeout)
        body = resp.read().decode("utf-8", errors="replace")
        try:
            return resp.status, json.loads(body)
        except json.JSONDecodeError:
            return resp.status, body
    except urllib.error.HTTPError as e:
        try:
            body = e.read().decode("utf-8", errors="replace")
        except Exception:
            body = ""
        return e.code, body
    except Exception as e:
        return 0, str(e)


def poll_anthropic(oauth_file: Path) -> PollResult:
    email = oauth_file.name.replace("claude-", "").replace(".json", "")
    try:
        d = json.loads(oauth_file.read_text())
    except Exception as e:
        return PollResult(
            account=email, provider="claude", status="error", error="parse_oauth: " + str(e)[:80]
        )
    token = d.get("access_token")
    if not token:
        return PollResult(account=email, provider="claude", status="error", error="no_access_token")
    headers = {
        "Authorization": "Bearer " + token,
        "anthropic-beta": ANTHROPIC_BETA_HEADER,
        "Content-Type": "application/json",
    }
    code, body = http_get_json(ANTHROPIC_USAGE_URL, headers)
    if code == 401:
        return PollResult(account=email, provider="claude", status="401", error="needs_refresh")
    if code == 403:
        return PollResult(account=email, provider="claude", status="403", error="needs_reauth")
    if code != 200 or not isinstance(body, dict):
        return PollResult(
            account=email, provider="claude", status=str(code or "error"),
            error=str(body)[:120] if body else "unknown",
        )
    five = body.get("five_hour") or {}
    seven = body.get("seven_day") or {}
    extra = body.get("extra_usage") or {}
    return PollResult(
        account=email,
        provider="claude",
        status="ok",
        primary_used_percent=five.get("utilization"),
        primary_reset_at=five.get("resets_at"),
        primary_limit_seconds=18000,  # 5h fixed window
        secondary_used_percent=seven.get("utilization"),
        secondary_reset_at=seven.get("resets_at"),
        secondary_limit_seconds=604800,  # 7d fixed window
        plan_type=None,  # Anthropic doesn't expose plan_type at this endpoint
        limit_reached=(five.get("utilization") or 0) >= 100 or (seven.get("utilization") or 0) >= 100,
        extra_usage_paid_pct=extra.get("utilization") if extra.get("is_enabled") else None,
        raw=body,
    )

def update_cpa_sqlite(db_path: Path, results: list[PollResult], dry_run: bool = False) -> dict:
    if not db_path.exists():
        return {"error": "db_missing", "path": str(db_path)}
    stats = {"updated": 0, "skipped": 0, "errors": []}
    if dry_run:
        for r in results:
            if r.status == "ok":
                stats["updated"] += 1
            else:
                stats["skipped"] += 1
        stats["dry_run"] = True
        return stats
    conn = sqlite3.connect(str(db_path))
    try:
        for r in results:
            if r.status != "ok":
                stats["skipped"] += 1
                continue
            email_key = r.account.split(" (")[0]
            try:
                conn.execute(
                    """
                    UPDATE usage_identities SET
                        primary_window_used_percent = ?,
                        primary_window_limit_seconds = ?,
                        primary_window_reset_at = ?,
                        secondary_window_used_percent = ?,
                        secondary_window_limit_seconds = ?,
                        secondary_window_reset_at = ?,
                        plan_type = ?,
                        limit_reached = ?,
                        stats_updated_at = ?
                    WHERE name = ? AND provider = ?
                    """,
                    (
                        int(r.primary_used_percent) if r.primary_used_percent is not None else None,
                        r.primary_limit_seconds,
                        r.primary_reset_at,
                        int(r.secondary_used_percent) if r.secondary_used_percent is not None else None,
                        r.secondary_limit_seconds,
                        r.secondary_reset_at,
                        r.plan_type,
                        1 if r.limit_reached else 0 if r.limit_reached is not None else None,
                        datetime.now(timezone.utc).isoformat(),
                        email_key,
                        r.provider,
                    ),
                )
                stats["updated"] += 1
            except Exception as e:
                stats["errors"].append({"account": r.account, "error": str(e)[:80]})
        conn.commit()
    finally:
        conn.close()
    return stats


def append_jsonl(jsonl_path: Path, records: list[dict]) -> None:
    jsonl_path.parent.mkdir(parents=True, exist_ok=True)
    with jsonl_path.open("a", encoding="utf-8") as f:
        for rec in records:
            f.write(json.dumps(rec, default=str) + "\n")


def poll_once(oauth_dir: Path, db: Path, jsonl: Path, dry_run: bool) -> dict:
    results = []
    for f in sorted(oauth_dir.glob("claude-*.json")):
        results.append(poll_anthropic(f))
    codex = oauth_dir / "codex-zfan7@sva.edu-pro.json"
    if codex.exists():
        results.append(poll_codex(codex))

    update_stats = update_cpa_sqlite(db, results, dry_run=dry_run)
    now = datetime.now(timezone.utc).isoformat()
    records = []
    for r in results:
        records.append({
            "ts": now,
            "account": r.account,
            "provider": r.provider,
            "status": r.status,
            "primary_used_percent": r.primary_used_percent,
            "primary_reset_at": r.primary_reset_at,
            "secondary_used_percent": r.secondary_used_percent,
            "secondary_reset_at": r.secondary_reset_at,
            "plan_type": r.plan_type,
            "limit_reached": r.limit_reached,
            "extra_usage_paid_pct": r.extra_usage_paid_pct,
            "error": r.error,
        })
    append_jsonl(jsonl, records)
    summary = {
        "ts": now,
        "polled": len(results),
        "ok": sum(1 for r in results if r.status == "ok"),
        "errors": [r for r in records if r["status"] != "ok"],
        "update_stats": update_stats,
        "dry_run": dry_run,
    }
    return summary


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--once", action="store_true", help="Run a single poll and exit")
    parser.add_argument("--start", action="store_true", help="Run cron loop")
    parser.add_argument("--stop", action="store_true", help="Send SIGTERM to running poller via PID file")
    parser.add_argument("--dry-run", action="store_true", help="Read but do not UPDATE SQLite")
    parser.add_argument("--interval-seconds", type=int, default=60, help="Cron interval (default 60s)")
    parser.add_argument("--oauth-dir", type=Path, default=DEFAULT_OAUTH_DIR)
    parser.add_argument("--db", type=Path, default=DEFAULT_CPA_DB)
    parser.add_argument("--jsonl", type=Path, default=DEFAULT_JSONL)
    parser.add_argument("--pid-file", type=Path, default=DEFAULT_PID_FILE)
    args = parser.parse_args()

    logging.basicConfig(
        level=logging.INFO,
        format="%(asctime)s [%(levelname)s] %(message)s",
    )

    if args.stop:
        if args.pid_file.exists():
            try:
                pid = int(args.pid_file.read_text().strip())
                import os
                import signal
                os.kill(pid, signal.SIGTERM)
                args.pid_file.unlink()
                LOGGER.info("Sent SIGTERM to PID %s", pid)
                return 0
            except Exception as e:
                LOGGER.error("stop failed: %s", e)
                return 1
        else:
            LOGGER.warning("PID file missing; nothing to stop")
            return 0

    if args.once:
        summary = poll_once(args.oauth_dir, args.db, args.jsonl, args.dry_run)
        print(json.dumps(summary, indent=2, default=str))
        return 0

    if args.start:
        args.pid_file.parent.mkdir(parents=True, exist_ok=True)
        import os
        args.pid_file.write_text(str(os.getpid()))
        LOGGER.info(
            "Aperant poller started; interval=%ds dry_run=%s",
            args.interval_seconds, args.dry_run,
        )
        try:
            while True:
                try:
                    summary = poll_once(args.oauth_dir, args.db, args.jsonl, args.dry_run)
                    LOGGER.info(
                        "poll: ok=%d/%d updated=%d",
                        summary["ok"], summary["polled"],
                        summary["update_stats"].get("updated", 0),
                    )
                except Exception as e:
                    LOGGER.error("poll iteration failed: %s", e)
                time.sleep(max(15, args.interval_seconds))
        except KeyboardInterrupt:
            LOGGER.info("Interrupted; cleaning up PID file")
        finally:
            try:
                args.pid_file.unlink()
            except FileNotFoundError:
                pass
        return 0

    parser.print_help()
    return 1


if __name__ == "__main__":
    sys.exit(main())

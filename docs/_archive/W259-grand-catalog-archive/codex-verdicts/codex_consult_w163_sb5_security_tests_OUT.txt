    """Return (subcommand, offset-after-subcommand) or None.

    Picks the subcommand nearest to 'git' so `git push origin commit` does not
    misclassify 'commit' as the subcommand.
    """
    start = 0
    while start < len(s):
        git_hit = _find_git(s, start)
        if not git_hit:
            return None
        git_idx, git_len = git_hit

        if _is_in_comment(s, git_idx):
            start = git_idx + git_len
            continue

        best_cmd: str | None = None
        best_idx = len(s) + 1

        for cmd in _GIT_COMMANDS_WITH_NO_VERIFY:
            search_pos = git_idx + git_len
            while search_pos < len(s):
                cmd_idx = s.find(cmd, search_pos)
                if cmd_idx == -1:
                    break
                before = s[cmd_idx - 1] if cmd_idx > 0 else " "
                after = s[cmd_idx + len(cmd)] if cmd_idx + len(cmd) < len(s) else " "
                if not _RE_WS.match(before):
                    search_pos = cmd_idx + 1
                    continue
                if after != "" and not _RE_SUBCMD_AFTER_CHAR.match(after):
                    search_pos = cmd_idx + 1
                    continue
                if _RE_SEMI_OR_PIPE.search(s[git_idx + git_len : cmd_idx]):
                    break
                if _is_in_comment(s, cmd_idx):
                    search_pos = cmd_idx + 1
                    continue

                # verify tokens between 'git' and the candidate are only flags or flag-arguments
                gap = s[git_idx + git_len : cmd_idx]
                tokens = [t for t in gap.strip().split() if t]
                only_flags_and_args = True
                expect_flag_arg = False
                for t in tokens:
                    if expect_flag_arg:
                        expect_flag_arg = False
                        continue
                    if t.startswith("-"):
                        # -c / -C / --work-tree / --git-dir / --namespace / --super-prefix take a following arg
                        if t in ("-c", "-C", "--work-tree", "--git-dir", "--namespace", "--super-prefix"):
                            expect_flag_arg = True
                        continue
                    only_flags_and_args = False
                    break
                if not only_flags_and_args:
                    search_pos = cmd_idx + 1
                    continue

                if cmd_idx < best_idx:
                    best_idx = cmd_idx
                    best_cmd = cmd
                break

        if best_cmd is not None:
            return (best_cmd, best_idx + len(best_cmd))
        start = git_idx + git_len
    return None

def _extract_command(raw: str) -> str:
    """Pull command from JSON payload or return raw."""
    trimmed = raw.strip()
    if not trimmed.startswith("{"):
        return trimmed
    try:
        parsed = json.loads(trimmed)
    except json.JSONDecodeError:
        return trimmed
    if not isinstance(parsed, dict):
        return trimmed
    tool_input = parsed.get("tool_input")
    if isinstance(tool_input, dict):
        cmd = tool_input.get("command")
        if isinstance(cmd, str):
            return cmd
    for key in ("command", "cmd", "input", "shell", "script"):
        val = parsed.get(key)
        if isinstance(val, str):
            return val
    return trimmed

def _split_segments(cmd: str) -> list[str]:
    """Split a Bash string by UNQUOTED command separators into individual command segments.

    Implements the same per-segment-validation contract that the removed
    bash_command_allowlist.py:split_command_segments helper used to provide before
    Wave 11A removal. Each segment is a single command + its args. Compound commands
    (a && b ; c) are split into [a, b, c] for individual validation.

    Quote-aware: separators inside single ('...') or double ("...") quotes are NOT
    split. This is critical for `python -c 'import sys; print(sys.version)'` which
    contains a literal ';' inside the quoted argument.
    """
    segments: list[str] = []
    current: list[str] = []
    i = 0
    in_single = False
    in_double = False
    while i < len(cmd):
        c = cmd[i]
        # Quote state tracking
        if c == "'" and not in_double:
            in_single = not in_single
            current.append(c)
            i += 1
            continue
        if c == '"' and not in_single:
            in_double = not in_double
            current.append(c)
            i += 1
            continue
        # Only split on separators OUTSIDE quotes
        if not in_single and not in_double:
            # Multi-char separators first (&& ||)
            if cmd[i : i + 2] in ("&&", "||"):
                segments.append("".join(current).strip())
                current = []
                i += 2
                continue
            # Single-char separators (; | &)
            # `&` is bash background-execution separator: `cmd1 & cmd2` runs cmd1 in
            # background AND cmd2 in foreground. Without splitting, leading-token
            # allowlist on cmd1 lets cmd2 (potentially destructive) execute unchecked.
            # Codex T1 F-004 conf=0.91 HIGH: `git status & rm -rf /` was accepted as
            # single segment with `git` allowlisted. Multi-char `&&` handled at L266-270
            # ABOVE this single-char check (precedence preserved by ordering).
            if c in (";", "|", "&"):
                segments.append("".join(current).strip())
                current = []
                i += 1
                continue
        current.append(c)
        i += 1
    # Append final segment
    tail = "".join(current).strip()
    if tail:
        segments.append(tail)
    return [s for s in segments if s]


def _first_token(segment: str) -> str:
    """Extract the first token (command name) from a segment, stripping basename + extension."""
    try:
        tokens = shlex.split(segment)
    except ValueError:
        # shlex can't parse (unmatched quote etc) — return raw split as fallback
        tokens = segment.split()
    if not tokens:
        return ""
    # Get basename without extension (e.g. /usr/bin/python3.11 -> python)
    cmd = os.path.basename(tokens[0]).lower()
    # Strip common executable extensions
    for ext in (".exe", ".cmd", ".bat", ".ps1"):
        if cmd.endswith(ext):
            cmd = cmd[: -len(ext)]
            break
    return cmd


def _validate_git_ambiguous_subcmd(subcmd: str, tokens: list[str], i: int) -> tuple[bool, str]:
    """Validate ambiguous git subcommands (branch/tag/config/remote/stash) by inspecting flags.

    Bare invocation (no flag args) defaults to READ for these (they list/show by default).
    Specific write flags trigger DENY.
    """
    next_arg = tokens[i + 1] if i + 1 < len(tokens) else ""

    if subcmd == "stash":
        # bare `git stash` = `git stash push` (write); list/show = read
        if next_arg in {"list", "show"}:
            return True, "allow:git stash list/show is read-only"
        return False, f"block:git stash {next_arg or '(bare = push)'} is write"

    if subcmd == "tag":
        if next_arg in {"-l", "--list", ""}:
            return True, "allow:git tag --list/bare is read-only"
        if next_arg in {"-d", "-D", "--delete"}:
            return False, "block:git tag --delete is write"
        # Creating a tag (e.g. `git tag v1.0`) is write
        return False, f"block:git tag {next_arg} (create) is write"

    if subcmd == "config":
        if next_arg in {"--get", "--list", "-l", "--get-all", "--get-regexp", "--show-origin"}:
            return True, "allow:git config --get/--list is read-only"
        if next_arg in {"--unset", "--unset-all", "--remove-section", "--rename-section"}:
            return False, "block:git config --unset/--remove is write"
        # Bare `git config foo.bar` (read single) is allowed; `git config foo.bar value` (set) is write
        # Heuristic: 1 positional after subcmd = read; 2+ positionals = write
        positionals = [t for t in tokens[i + 1 :] if not t.startswith("-")]
        if len(positionals) <= 1:
            return True, "allow:git config single positional is read"
        return False, "block:git config with value (set) is write"

    if subcmd == "branch":
        if next_arg in {"", "-l", "--list", "-a", "-r", "--all", "--remotes", "-v", "-vv", "--verbose"}:
            return True, "allow:git branch list/bare is read-only"
        if next_arg in {"-d", "-D", "--delete"}:
            return False, "block:git branch --delete is write"
        if next_arg in {"-m", "-M", "--move"}:
            return False, "block:git branch --move is write"
        if next_arg in {"-c", "-C", "--copy"}:
            return False, "block:git branch --copy is write"
        # Creating a branch (e.g. `git branch new-feature`) is write
        return False, f"block:git branch {next_arg} (create) is write"

    if subcmd == "remote":
        if next_arg in {"", "-v", "--verbose", "show", "get-url"}:
            return True, "allow:git remote bare/-v/show is read-only"
        if next_arg in {"add", "remove", "rename", "set-url", "set-head", "prune", "rm", "set-branches"}:
            return False, f"block:git remote {next_arg} is write"
        return False, f"block:git remote {next_arg} unknown subcommand (fail-closed)"

    return False, f"block:git {subcmd} ambiguous subcommand fall-through (should not reach)"


def _validate_git(segment: str) -> tuple[bool, str]:
    """Validate `git <subcmd>` — allow read subcommands, deny write subcommands."""
    try:
        tokens = shlex.split(segment)
    except ValueError:
        return False, "block:git unparseable"
    # Skip `git` and any global flags like `-C path`, `--git-dir=...`, `--work-tree=...`
    i = 1
    while i < len(tokens):
        t = tokens[i]
        if t.startswith("--") or t == "-C":
            # Skip -C and its arg, or --flag=value
            i += 2 if t == "-C" else 1
            continue
        if t.startswith("-"):
            i += 1
            continue
        break
    if i >= len(tokens):
        return True, "allow:bare git command (e.g. `git`)"
    subcmd = tokens[i].lower()
    # Ambiguous subcommands (read or write depending on flags) — check FIRST
    if subcmd in _GIT_AMBIGUOUS_SUBCMDS:
        return _validate_git_ambiguous_subcmd(subcmd, tokens, i)
    # Pure write subcommands
    if subcmd in _GIT_WRITE_SUBCMDS:
        return False, f"block:git {subcmd} is write subcommand"
    # Pure read subcommands
    if subcmd in _GIT_READ_SUBCMDS:
        return True, f"allow:git {subcmd} is read-only"
    # Unknown subcommand — fail closed (per security-checklist.md fail-closed posture)
    return False, f"block:git {subcmd} unknown subcommand (fail-closed)"


def _validate_codex(segment: str, policy: str) -> tuple[bool, str]:
    """Validate `codex ...` invocation per policy mode (token-based for boundary safety).

    Regex `\\b--word\\b` fails to match `--word` because `\\b` requires word/non-word
    transition; `--` is non-word so no boundary triggers. Token-based check via shlex.split
    is the SOTA-recommended fix per security-checklist allowlist patterns.
    """
    if policy != "codex-readonly":
        return False, "block:codex invocation requires --policy codex-readonly"
    try:
        tokens = shlex.split(segment)
    except ValueError:
        return False, "block:codex unparseable"

    # Token-based deny check (closes \b regex boundary issue)
    for tok in tokens:
        if tok in _CODEX_DENY_TOKENS:
            return False, f"block:codex deny-token {tok}"
        # Prefix check for attached/equals forms (codex T1 F-003 residual hardening):
        # `--config=key=value` and `-cKEY=VALUE` bypass exact-token deny above.
        # Skip bare exact matches already in DENY_TOKENS to avoid duplicate-error noise.
        for prefix in _CODEX_DENY_TOKEN_PREFIXES:
            if tok.startswith(prefix) and tok != prefix and tok not in _CODEX_DENY_TOKENS:
                return False, f"block:codex deny-prefix {prefix} in token {tok}"
        # Substring check for embedded patterns (e.g. -p deep-review-exec)
        for sub in _CODEX_DENY_SUBSTRINGS:
            if sub in tok.lower():
                return False, f"block:codex deny-substring {sub} in token {tok}"

    # Structural allow check: require specific subcommand shape
    # Acceptable forms:
    #   codex --version | -V
    #   codex --help | -h
    #   codex exec --help | -h
    #   codex exec [--ephemeral] -p deep-review [...]            (+ allowed extras)
    #   codex exec review --uncommitted [--ephemeral] [...]
    if len(tokens) < 2:
        return False, "block:bare codex invocation"

    # `codex --version` / `codex --help`
    if tokens[1] in {"--version", "-V", "--help", "-h"}:
        return True, f"allow:codex {tokens[1]}"

    # All allowed forms after `codex` start with `exec`
    if tokens[1] != "exec":
        return False, f"block:codex subcommand '{tokens[1]}' not in readonly allowlist"

    # `codex exec --help` / `codex exec -h`
    if len(tokens) >= 3 and tokens[2] in {"--help", "-h"}:
        return True, "allow:codex exec --help"

    # `codex exec review --uncommitted ...` (review subcommand)
    if len(tokens) >= 3 and tokens[2] == "review":
        if "--uncommitted" not in tokens:
            return False, "block:codex exec review requires --uncommitted in readonly mode"
        # All tokens after `review` must be in safe set
        return True, "allow:codex exec review --uncommitted"

    # `codex exec [...] -p deep-review [...]` — REQUIRES explicit -p deep-review
    # Sandbox-escalation closure (Stop hook 2026-05-03):
    # - `codex exec --ephemeral` WITHOUT -p was a bypass: default profile is operator-
    #   configurable and could be `deep-review-exec` (danger-full-access).
    # - `codex exec -p deep-review --sandbox X` overrides profile sandbox; closed by
    #   _CODEX_DENY_TOKENS containing --sandbox.
    # - `codex exec -p deep-review -c sandbox.mode=X` overrides via inline config;
    #   closed by _CODEX_DENY_TOKENS containing -c / --config.
    # The strict requirement: explicit `-p deep-review` is the ONLY allowed exec form.
    if "-p" not in tokens:
        return False, "block:codex exec requires explicit -p deep-review (default profile is operator-configurable, not safe to assume readonly)"
    p_idx = tokens.index("-p")
    if p_idx + 1 >= len(tokens):
        return False, "block:codex -p missing profile name"
    profile = tokens[p_idx + 1]
    if profile != "deep-review":
        return False, f"block:codex -p {profile} not in readonly allowlist (only deep-review)"
    return True, "allow:codex exec -p deep-review"


def _validate_sed(segment: str) -> tuple[bool, str]:
    """sed -i / --in-place is write; sed `e` flag/command spawns subprocess; -n print is read."""
    for pat in _SED_DENY_PATTERNS:
        if pat.search(segment):
            # Distinguish reasons for forensic clarity
            ps = pat.pattern
            if "in-place" in ps or r"-i\b" in ps:
                return False, "block:sed -i/--in-place writes to file"
            if "e[gimpI" in ps or "0-9]*e" in ps:
                return False, "block:sed s///e flag executes replacement as shell (subprocess spawn)"
            if r"e\s+\S" in ps:
                return False, "block:sed `e` command evaluates pattern space as shell (subprocess spawn)"
            return False, f"block:sed deny pattern ({ps})"
    return True, "allow:sed without -i/--in-place/e-flag is read-only"


def _validate_awk(segment: str) -> tuple[bool, str]:
    """awk subprocess-spawning is HARD DENY (system, pipe-to-cmd, pipe-from-cmd, getline-coproc).

    Stop-hook fix-forward round-2 2026-05-03 — awk's Turing-complete language has
    multiple subprocess primitives that bypass the leading-verb allowlist. Read-only
    mode rejects them all; legitimate read-only awk uses print/printf/match/etc.
    without subprocess spawning. Also rejects `-f script.awk` (uninspectable file).
    """
    # Tokenize for flag inspection
    try:
        tokens = shlex.split(segment)
    except ValueError:
        # Unparseable — fail-closed (likely contains injected metacharacters)
        return False, "block:awk segment unparseable (fail-closed)"

    # Reject awk option flags that bypass script inspection (read script from file)
    for tok in tokens[1:]:
        # Match flag itself or flag=value form (e.g. --file=script.awk)
        flag = tok.split("=", 1)[0] if "=" in tok else tok
        if flag in _AWK_DENY_FLAGS:
            return False, (
                f"block:awk {tok} reads script from external source (uninspectable); "
                "read-only mode requires inline script literal"
            )

    # Reject subprocess-spawning patterns inside awk script string
    for pat in _AWK_SUBPROCESS_DENY_PATTERNS:
        if pat.search(segment):
            return False, (
                f"block:awk subprocess pattern ({pat.pattern}) — awk system()/pipe-to/"
                "pipe-from/coprocess/extension-load bypass leading-verb allowlist"
            )

    return True, "allow:awk without system()/pipe/coprocess is read-only"


def _validate_find(segment: str) -> tuple[bool, str]:
    """find with -delete / -exec / -execdir is write; bare find or find -type is read."""
    if re.search(r"-delete\b", segment):
        return False, "block:find -delete is write"
    if re.search(r"-exec(?:dir)?\b", segment):
        return False, "block:find -exec/-execdir spawns subprocesses (deny by default)"
    if re.search(r"-fprint\b", segment):
        return False, "block:find -fprint is write"
    return True, "allow:find without -delete/-exec is read-only"


def _validate_python(segment: str) -> tuple[bool, str]:
    """python -c with limited surface; python script.py blocked (could be anything)."""
    try:
        tokens = shlex.split(segment)
    except ValueError:
        return False, "block:python unparseable"
    if len(tokens) < 2:
        return False, "block:bare python invocation (interactive REPL not supported in subagent)"
    # Allow python --version / python -V / python -h / python --help
    if tokens[1] in {"--version", "-V", "-h", "--help"}:
        return True, f"allow:python {tokens[1]} is read-only"
    # Block python -m pip install / etc.
    if tokens[1] == "-m" and len(tokens) > 2 and tokens[2] in {"pip", "ensurepip", "venv"}:
        return False, f"block:python -m {tokens[2]} is package management"
    # Allow python -c with no obvious mutation keywords
    if tokens[1] == "-c" and len(tokens) > 2:
        code = tokens[2]
        # Conservative deny of mutation patterns
        deny_patterns = ("open(", "os.system", "os.remove", "os.unlink", "os.rmdir",
                          "shutil.rm", "subprocess", "Popen", "exec(", "eval(",
                          "import os.system", "__import__", "compile(")
        for dp in deny_patterns:
            if dp in code:
                return False, f"block:python -c contains mutation pattern ({dp})"
        return True, "allow:python -c with no mutation patterns"
    return False, "block:python with unknown args (fail-closed)"


def _validate_verifier_pytest(segment: str) -> tuple[bool, str]:
    """Verifier policy: allow `pytest` with safe args; deny `-c` (Python eval) + bad --basetemp."""
    try:
        tokens = shlex.split(segment, posix=False)
    except ValueError:
        return False, "block:pytest command parsing failed"
    for i, t in enumerate(tokens):
        if t == "-c":
            return False, "block:pytest -c (arbitrary Python eval) denied under verifier policy"
        # --basetemp escape: deny ALL paths that escape cwd, not just absolute paths.
        # Three bypass classes closed (Stop hook block #5, 2026-05-03):
        #   (1) Absolute paths: /tmp/evil, \evil, C:\evil — original check
        #   (2) Tilde expansion: ~/evil, ~user/evil — shell expands to abs before pytest sees
        #   (3) Path traversal: ../../../tmp/evil, ./../../tmp, subdir/../../tmp —
        #       relative paths whose normpath escapes cwd via `..` segments
        # Verifier writes only to worktree-relative test session dirs that DESCEND from cwd.
        bt = None
        if t == "--basetemp" and i + 1 < len(tokens):
            bt = tokens[i + 1]
        elif t.startswith("--basetemp="):
            bt = t.split("=", 1)[1]
        if bt:
            # Stop hook block #5 fix-forward (2026-05-03): strip outer matched quotes.
            # shlex.split(posix=False) at L589 PRESERVES literal quote chars in tokens,
            # so `pytest --basetemp "/tmp/evil"` yields bt=`"/tmp/evil"` (with quotes).
            # The absolute-path check at L611 sees first char `"` not `/` and bypasses.
            # Strip outer matched single/double quotes BEFORE path-class checks.
            if len(bt) >= 2 and bt[0] == bt[-1] and bt[0] in ('"', "'"):
                bt = bt[1:-1]
            # Also strip equals-form bt that may carry quotes after =, e.g. --basetemp="..."
            # (handled by the same outer-quote strip above)
            # (1) Absolute paths (Unix `/`, Windows `\` or drive-letter `C:`)
            if bt.startswith("/") or bt.startswith("\\") or (len(bt) > 1 and bt[1] == ":"):
                return False, f"block:pytest --basetemp must be cwd-relative; absolute path rejected ({bt})"
            # (2) Tilde expansion (shell expands ~/X and ~user/X to absolute)
            if bt.startswith("~"):
                return False, f"block:pytest --basetemp must not use `~` tilde expansion; rejected ({bt})"
            # (2b) Shell-expansion sigils — bash WOULD expand $HOME, ${HOME}, $PWD, $TMPDIR,
            #      %USERPROFILE%, `cmd` to absolute paths BEFORE pytest receives the arg.
            #      Codex T1 NEEDS-REVISION conf=0.91 F-002 P1 verdict
            #      (.claude/state/codex_consult_pytest_basetemp_traversal_fix_OUT.txt:2988-2990)
            #      LIVE PROBE: `pytest --basetemp $HOME/evil => allowed=True` pre-fix.
            #      Global denylist at L236 catches `${...}`, `$(`, backticks BUT NOT bare `$HOME`.
            if any(sigil in bt for sigil in ("$", "%", "`")):
                return False, f"block:pytest --basetemp must not use shell-expansion sigils ($,%,`); rejected ({bt})"
            # (3) Path traversal via `..` segments — normpath resolves them; if result is
            #     `..` itself OR starts with `../`, the original path escapes cwd. Sanitize
            #     Windows `\` to `/` for cross-platform consistency. Codex T1 Q2 prescription
            #     (.claude/state/codex_consult_pytest_basetemp_traversal_fix_OUT.txt:3045):
            #     prefer `== ".." or startswith("../")` over `startswith("..")` to avoid
            #     over-denying legit dirnames like `..safe/foo` (which normpath preserves).
            bt_norm = os.path.normpath(bt).replace("\\", "/")
            if bt_norm == ".." or bt_norm.startswith("../"):
                return False, f"block:pytest --basetemp must stay within cwd; `..` traversal rejected ({bt} -> {bt_norm})"
    return True, "allow:pytest with safe args (verifier policy)"

def _validate_segment(segment: str, policy: str) -> tuple[bool, str]:
    """Validate a single command segment per policy."""
    if not segment.strip():
        return True, "allow:empty segment"
    cmd = _first_token(segment)
    if not cmd:
        return False, "block:could not extract command name"

    # Verifier-policy pre-check for verification tools.
    # Must run BEFORE _ALWAYS_DENY because npm/cargo/go are in always-deny but
    # verifier policy permits specific subcommands (npm test, cargo test/check, go test).
    if policy == "verifier":
        if cmd == "pytest":
            return _validate_verifier_pytest(segment)
        if cmd == "pyright":
            return _validate_verifier_pyright(segment)
        if cmd == "ruff":
            return _validate_verifier_ruff(segment)
        if cmd == "mypy":
            return _validate_verifier_mypy(segment)
        if cmd == "tsc":
            return _validate_verifier_tsc(segment)
        if cmd == "npm":
            return _validate_verifier_npm(segment)
        if cmd == "cargo":
            return _validate_verifier_cargo(segment)
        if cmd == "go":
            return _validate_verifier_go(segment)

    # Always-deny check (defense-in-depth for non-verifier policies AND
    # for commands not handled by verifier-specific validators)
    if cmd in _ALWAYS_DENY:
        return False, f"block:{cmd} is in always-deny set"

    # Per-command validators
    if cmd == "git":
        return _validate_git(segment)
    if cmd == "codex":
        return _validate_codex(segment, policy)
    if cmd == "sed":
        return _validate_sed(segment)
    if cmd == "find":
        return _validate_find(segment)
    if cmd == "awk":
        return _validate_awk(segment)
    if cmd in {"python", "python3"}:
        return _validate_python(segment)

    # Generic readonly allowlist check (verifier inherits readonly allowlist as superset)
    if cmd in _READONLY_COMMANDS:
        return True, f"allow:{cmd} is in readonly allowlist"

    # Unknown command — fail closed
    return False, f"block:{cmd} not in {policy} allowlist (fail-closed)"


def validate_command(cmd: str, policy: str = "readonly") -> tuple[bool, str]:
    """Sync test-friendly entry point: validate Bash command string against policy.

    Returns (allowed: bool, reason: str). Reason prefixed with "allow:" or "block:".
    """
    # Stop-hook fix-forward 2026-05-03 — FAIL-CLOSED on shell substitutions
    # BEFORE any segment validation. Substitutions evaluate at exec time and
    # bypass leading-verb allowlists (e.g. `git diff $(rm -rf /)` allowlists
    # `git diff` but inner command runs at substitution time). Probe at
    # tmp/probe_substitution_bypass.py — n=6 distinct bypass shapes confirmed
    # under prior code; all closed by this hard-deny.
    for pat in _SHELL_SUBSTITUTION_DENY_PATTERNS:
        if pat.search(cmd):
            return False, (
                f"block:shell substitution detected ({pat.pattern}) — "
                "read-only mode rejects substitutions; inner command bypasses "
                "leading-verb allowlist (Stop-hook 2026-05-03 fix-forward)"
            )

    # Newline as command separator — _COMMAND_SEPARATORS regex misses it.
    # Multiline commands sneak destructive segments past per-segment validation.

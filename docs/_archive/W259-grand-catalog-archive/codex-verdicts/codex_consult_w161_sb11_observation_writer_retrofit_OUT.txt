            current_mode = stat.S_IMODE(os.fstat(fd).st_mode)
            if current_mode != mode:
                file_uid = os.fstat(fd).st_uid
                if file_uid == os.getuid():
                    # Round-7 fchmod raises propagate (PermissionError, EROFS,
                    # alien-owner errno) — caller sees the failure, no silent-pass.
                    os.fchmod(fd, mode)
                # alien-owner case: leave perms intact; operator runs migration
                # command per rule body §Migration semantic for legacy files

        # Round-3-4: take exclusive advisory lock; Round-5 F-2 fail-closed
        # Round-7b: lock_fd captures Windows separate-lockfile fd (None on POSIX)
        lock_fd = _acquire_lock(fd, path)

        try:
            # Round-5 F-3: json encode raises propagate (no silent-drop)
            line = json.dumps(record, ensure_ascii=False) + "\n"
            encoded_bytes = line.encode("utf-8")

            # Round-5 F-5: byte-count check (NOT key count of the dict)
            if len(encoded_bytes) > PIPE_BUF_BYTES:
                raise RecordTooLarge(
                    f"Encoded record is {len(encoded_bytes)} bytes; "
                    f"exceeds platform record byte bound {PIPE_BUF_BYTES} bytes. "
                    f"Reject record OR shard via filename-namespace per Round-5 F-5."
                )

            # Round-3 atomic write: low-level os.write under lock
            # Never use buffered text writes (f.write splits into multiple syscalls)
            bytes_written = os.write(fd, encoded_bytes)
            if bytes_written != len(encoded_bytes):
                # Short-write defense: rare under O_APPEND but possible
                raise OSError(
                    errno.EIO,
                    f"Short write: {bytes_written} of {len(encoded_bytes)} bytes",
                )

            # Round-3-5 F-4: fsync if durable contracted; raises propagate
            if durable:
                os.fsync(fd)

            return bytes_written
        finally:
            _release_lock(fd, lock_fd)
    finally:
        os.close(fd)

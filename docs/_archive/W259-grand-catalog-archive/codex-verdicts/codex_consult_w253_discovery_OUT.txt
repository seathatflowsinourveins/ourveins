def parse_last_page(link):
    if not link: return None
    for part in link.split(','):
        if 'rel=\"last\"' in part:
            m=re.search(r'[?&]page=(\\d+)', part)
            if m: return int(m.group(1))
    return None

def velocity_score(full, stars):
    if stars == 0: return 0, 0

    last=max(1, math.ceil(stars/per))

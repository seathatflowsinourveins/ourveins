def parse_iso8601(s: Optional[str]) -> Optional[datetime]:
    if not s:
        return None
    try:
        if s.endswith("Z"):
            s = s[:-1] + "+00:00"
        return datetime.fromisoformat(s)
    except (ValueError, TypeError):
        return None


def fetch_fleet_state(db_path: Path) -> dict:
    if not db_path.exists():
        return {}
    state: dict = {}
    conn = sqlite3.connect(str(db_path))
    conn.row_factory = sqlite3.Row
    try:
        for row in conn.execute(
            "SELECT name, provider, primary_window_used_percent, "
            "primary_window_reset_at, secondary_window_used_percent, "
            "secondary_window_reset_at, plan_type, limit_reached, last_used_at "
            "FROM usage_identities WHERE provider IN ('claude', 'codex')"
        ):
            d = dict(row)
            state[d["name"] + "|" + d["provider"]] = d
    finally:
        conn.close()
    return state


def find_oauth_file(oauth_dir: Path, name: str, provider: str) -> Optional[Path]:
    if provider == "claude":
        candidate = oauth_dir / ("claude-" + name + ".json")
        if candidate.exists():
            return candidate
    elif provider == "codex":
        for f in oauth_dir.glob("codex-" + name + "*.json"):
            return f
    return None


def update_oauth_disabled(oauth_file: Path, new_disabled: bool, backup_dir: Path, dry_run: bool = False) -> dict:
    try:
        d = json.loads(oauth_file.read_text())
    except Exception as e:
        return {"file": oauth_file.name, "error": "parse: " + str(e)[:80]}

    current = d.get("disabled", False)
    if current == new_disabled:
        return {"file": oauth_file.name, "current": current, "new": new_disabled, "changed": False}

    if dry_run:
        return {"file": oauth_file.name, "current": current, "new": new_disabled, "changed": True, "dry_run": True}

    try:
        backup_dir.mkdir(parents=True, exist_ok=True)
        ts = datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%SZ")
        backup_path = backup_dir / (oauth_file.name + "." + ts + ".bak")
        shutil.copy2(oauth_file, backup_path)
    except Exception as e:
        return {"file": oauth_file.name, "error": "backup: " + str(e)[:80]}

    d["disabled"] = new_disabled
    d["disabled_updated_at"] = datetime.now(timezone.utc).isoformat()
    d["disabled_reason"] = "rotation_planner: 7d_threshold" if new_disabled else "rotation_planner: 7d_recovered"
    try:
        tmp = oauth_file.with_suffix(".tmp")
        tmp.write_text(json.dumps(d, indent=2))
        tmp.replace(oauth_file)
    except Exception as e:
        return {"file": oauth_file.name, "error": "write: " + str(e)[:80]}

    return {"file": oauth_file.name, "current": current, "new": new_disabled, "changed": True, "backup": str(backup_path)}


def plan_once(oauth_dir: Path, db: Path, jsonl: Path, backup_dir: Path, threshold_pct: int, recovery_buffer_pct: int, dry_run: bool) -> dict:
    state = fetch_fleet_state(db)
    if not state:
        return {"error": "no_fleet_state", "db": str(db)}

    now = datetime.now(timezone.utc)
    actions = []

    for key, row in state.items():
        name = row["name"]
        provider = row["provider"]
        f = find_oauth_file(oauth_dir, name, provider)
        if not f:
            actions.append({"key": key, "error": "oauth_file_not_found"})
            continue

        secondary_pct = row.get("secondary_window_used_percent")
        if secondary_pct is None:
            actions.append({"key": key, "status": "no_data", "skipped": True})
            continue

        secondary_pct = float(secondary_pct)
        secondary_reset = parse_iso8601(row.get("secondary_window_reset_at"))

        if secondary_pct >= threshold_pct:
            decision = True
            reason = "7d_used=" + str(secondary_pct) + "%>=" + str(threshold_pct) + "%"
        elif secondary_reset and secondary_reset <= now:
            decision = False
            reason = "7d_reset_past=" + secondary_reset.isoformat()
        elif secondary_pct < (threshold_pct - recovery_buffer_pct):
            decision = False
            reason = "7d_used=" + str(secondary_pct) + "%<" + str(threshold_pct - recovery_buffer_pct) + "%"
        else:
            try:
                cur_d = json.loads(f.read_text())
                decision = cur_d.get("disabled", False)
                reason = "hysteresis_band"
            except Exception:
                decision = False
                reason = "hysteresis_band_default_enabled"

        change = update_oauth_disabled(f, decision, backup_dir, dry_run=dry_run)
        change["key"] = key
        change["secondary_pct"] = secondary_pct
        change["reason"] = reason
        actions.append(change)

    jsonl.parent.mkdir(parents=True, exist_ok=True)
    with jsonl.open("a", encoding="utf-8") as fp:
        rec = {"ts": now.isoformat(), "threshold_pct": threshold_pct, "actions": actions, "dry_run": dry_run}
        fp.write(json.dumps(rec, default=str) + "\n")

    return {
        "ts": now.isoformat(),
        "fleet_size": len(state),
        "decisions_made": sum(1 for a in actions if a.get("changed")),
        "no_data": sum(1 for a in actions if a.get("status") == "no_data"),
        "errors": [a for a in actions if "error" in a],
        "actions": actions,
    }


def restore_from_backups(oauth_dir: Path, backup_dir: Path, dry_run: bool = False) -> dict:
    if not backup_dir.exists():
        return {"error": "no_backup_dir"}
    restored = []
    backups_by_name: dict = {}
    for b in backup_dir.glob("*.bak"):
        parts = b.name.split(".")
        if len(parts) < 3:
            continue
        json_idx = next((i for i, p in enumerate(parts) if p == "json"), None)
        if json_idx is None:
            continue
        orig = ".".join(parts[: json_idx + 1])
        backups_by_name.setdefault(orig, []).append(b)
    for orig, backups in backups_by_name.items():
        latest = sorted(backups)[-1]
        target = oauth_dir / orig
        if dry_run:
            restored.append({"file": orig, "backup": str(latest), "dry_run": True})
        else:
            try:
                shutil.copy2(latest, target)
                restored.append({"file": orig, "restored_from": str(latest)})
            except Exception as e:
                restored.append({"file": orig, "error": str(e)[:80]})
    return {"restored": restored, "dry_run": dry_run}

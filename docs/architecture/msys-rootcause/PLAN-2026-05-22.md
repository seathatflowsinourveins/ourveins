```javascript
#!/usr/bin/env node
// tools/precommit-msys-diag.mjs
// Phase 1 diagnostic: hashes tracked files before/after each pre-commit hook,
// reports content/mtime deltas, attributes to culprit hook.
// Read-only forensic tool. Exit 0 always.

import { createHash } from 'node:crypto';
import { readFileSync, statSync } from 'node:fs';
import { execFileSync } from 'node:child_process';
import { resolve } from 'node:path';

const args = Object.fromEntries(process.argv.slice(2).reduce((acc, a, i, arr) => {
  if (a.startsWith('--')) acc.push([a.slice(2), arr[i+1] && !arr[i+1].startsWith('--') ? arr[i+1] : true]);
  return acc;
}, []));

const HOOK = args.hook || null;
const TEE_FILE = args.tee || null;
const JSON_ONLY = args.json === true;

function listTrackedFiles() {
  const out = execFileSync('git', ['ls-files'], { encoding: 'utf8' });
  return out.trim().split('\n').filter(Boolean);
}

function snapshot(files) {
  const snap = new Map();
  for (const f of files) {
    try {
      const content = readFileSync(f);
      const hash = createHash('sha256').update(content).digest('hex');
      const mtime = statSync(f).mtimeMs;
      snap.set(f, { hash, mtime, size: content.length });
    } catch (e) { /* deleted/unreadable - skip */ }
  }
  return snap;
}

function diff(before, after) {
  const contentDelta = [], mtimeDelta = [];
  for (const [f, a] of after) {
    const b = before.get(f);
    if (!b) continue;
    if (b.hash !== a.hash) {
      const aContent = readFileSync(f);
      const bIdx = aContent.indexOf(0); // placeholder; we don't have before content
      contentDelta.push({ file: f, beforeHash: b.hash.slice(0,12), afterHash: a.hash.slice(0,12), sizeBefore: b.size, sizeAfter: a.size });
    } else if (b.mtime !== a.mtime) {
      mtimeDelta.push({ file: f, mtimeBefore: b.mtime, mtimeAfter: a.mtime });
    }
  }
  return { contentDelta, mtimeDelta };
}

function runHook(hookId) {
  try {
    execFileSync('pre-commit', ['run', hookId, '--verbose', '--all-files'], {
      encoding: 'utf8', stdio: ['ignore', 'pipe', 'pipe']
    });
  } catch (e) { /* hook may exit non-zero; we don't care here */ }
}

function main() {
  const files = listTrackedFiles();
  const before = snapshot(files);
  if (HOOK) runHook(HOOK);
  const after = snapshot(files);
  const { contentDelta, mtimeDelta } = diff(before, after);

  const report = {
    hook: HOOK || 'all',
    tracked_file_count: files.length,
    content_delta: contentDelta,
    mtime_delta: mtimeDelta,
    attribution_verdict: `${HOOK || 'pre-commit chain'} caused ${contentDelta.length} content + ${mtimeDelta.length} mtime deltas`
  };

  if (JSON_ONLY) {
    console.log(JSON.stringify(report, null, 2));
  } else {
    console.log(`┌─ Hook: ${report.hook} ───────┐`);
    console.log(`│ Tracked files: ${files.length}`);
    console.log(`├─ Content deltas (n=${contentDelta.length}): ─┐`);
    for (const d of contentDelta) console.log(`│   ${d.file}  ${d.beforeHash}→${d.afterHash}  ${d.sizeBefore}B→${d.sizeAfter}B`);
    console.log(`├─ mtime-only deltas (n=${mtimeDelta.length}): ─┐`);
    for (const d of mtimeDelta) console.log(`│   ${d.file}`);
    console.log(`└─ ATTRIBUTION: ${report.attribution_verdict}`);
  }

  if (TEE_FILE) {
    const fs = require('node:fs');
    fs.writeFileSync(TEE_FILE, JSON.stringify(report, null, 2));
  }

  process.exit(0); // forensic - never block
}

main();
```

```javascript
// tmp/msys-sandbox/run-bisect.mjs
// Runs the bisect matrix per SPEC Phase 2.
import { execFileSync } from 'node:child_process';
import { writeFileSync, readFileSync } from 'node:fs';

const MATRIX = [
  // [variable-name, [values...], setup-cmd-template]
  ['core.autocrlf', ['true', 'false', 'input'], (v) => ['git', 'config', 'core.autocrlf', v]],
  ['shell', ['bash', 'cmd'], null], // shell varied by env, not git config
  ['filetype', ['md', 'py', 'mix'], null], // varied by staged content
];

const results = [];

function testCommit(label) {
  writeFileSync('foo.md', `test ${label}\n`);
  execFileSync('git', ['add', 'foo.md']);
  try {
    const out = execFileSync('git', ['commit', '-m', `test: ${label}`, '--no-verify'], { encoding: 'utf8' });
    // Now run pre-commit manually to see the misfire
    const preOut = execFileSync('pre-commit', ['run', 'msys-hooks-form', '--all-files'], { encoding: 'utf8', stdio: ['ignore', 'pipe', 'pipe'] });
    return { label, outcome: 'PASS', preOut };
  } catch (e) {
    return { label, outcome: 'FAIL', err: String(e.stdout || e.stderr || e) };
  }
}

for (const [varName, values, setupTpl] of MATRIX) {
  for (const v of values) {
    if (setupTpl) execFileSync(...setupTpl(v));
    const r = testCommit(`${varName}=${v}`);
    results.push({ variable: varName, value: v, ...r });
    // reset for next cell
    execFileSync('git', ['reset', '--hard', 'HEAD~1'], { stdio: ['ignore', 'pipe', 'pipe'] });
  }
}

console.log(JSON.stringify(results, null, 2));
```

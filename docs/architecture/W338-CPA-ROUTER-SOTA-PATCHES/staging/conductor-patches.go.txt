// nextQuotaCooldown returns the next cooldown duration and updated backoff
// level for repeated quota errors.
//
// W338 Patch 2: full-jitter exponential backoff (AWS Brooker 2015).
// Range = [maxCooldown/4, maxCooldown] — base/4 prevents zero-wait,
// maxCooldown is the classic exponential ceiling for level N.
func nextQuotaCooldown(prevLevel int, disableCooling bool) (time.Duration, int) {
	if prevLevel < 0 {
		prevLevel = 0
	}
	if disableCooling {
		return 0, prevLevel
	}
	maxCooldown := quotaBackoffBase * time.Duration(1<<prevLevel)
	if maxCooldown < quotaBackoffBase {
		maxCooldown = quotaBackoffBase
	}
	if maxCooldown >= quotaBackoffMax {
		return quotaBackoffMax, prevLevel
	}
	jitterRange := int64(maxCooldown - maxCooldown/4)
	if jitterRange < 1 {
		jitterRange = 1
	}
	jittered := time.Duration(rand.Int63n(jitterRange)) + (maxCooldown / 4)
	return jittered, prevLevel + 1
}

// nextOverloadCooldown returns the next cooldown for HTTP 529 with full-jitter.
// W338 Patch 1: separate from quota backoff because 529 = provider capacity,
// not per-account quota. Avoids polluting the BackoffLevel ladder on
// upstream-side incidents the auth itself is not at fault for.
func nextOverloadCooldown(prevLevel int) (time.Duration, int) {
	if prevLevel < 0 {
		prevLevel = 0
	}
	maxCooldown := overloadBackoffBase * time.Duration(1<<prevLevel)
	if maxCooldown >= overloadBackoffMax {
		maxCooldown = overloadBackoffMax
	}
	jitterRange := int64(maxCooldown - maxCooldown/4)
	if jitterRange < 1 {
		jitterRange = 1
	}
	jittered := time.Duration(rand.Int63n(jitterRange)) + (maxCooldown / 4)
	if jittered >= overloadBackoffMax {
		return overloadBackoffMax, prevLevel
	}
	return jittered, prevLevel + 1
}

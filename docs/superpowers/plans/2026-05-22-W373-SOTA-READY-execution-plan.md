```python
"""Parse stream-output Markdown findings tables into Finding objects."""
from __future__ import annotations
import re
from typing import List
from .finding_models import Finding, RiskClass, RemediationType

class StreamParseError(ValueError):
    pass

_TABLE_HEADER_RE = re.compile(
    r"^\s*\|\s*ID\s*\|\s*\w+\s*\|.*\|\s*Risk-?class-?draft\s*\|.*$",
    re.IGNORECASE | re.MULTILINE,
)
_ROW_RE = re.compile(
    r"^\s*\|\s*(W373-[A-F]-F\d+)\s*\|"     # ID
    r"\s*([^|]+?)\s*\|"                     # category
    r"\s*([^|]+?)\s*\|"                     # subject
    r"\s*([^|]+?)\s*\|"                     # evidence
    r"\s*(LOW|HIGH|MED|MEDIUM)\s*\|"        # risk
    r"\s*([\d.]+|n/a)\s*\|",                # sca
    re.IGNORECASE | re.MULTILINE,
)

def _risk_to_class(s: str) -> RiskClass:
    s = s.upper().strip()
    if s in ("LOW",):
        return RiskClass.LOW
    if s in ("HIGH", "MED", "MEDIUM"):     # MED collapsed to HIGH per Q2 binary
        return RiskClass.HIGH
    raise StreamParseError(f"unknown risk-class: {s!r}")

def _infer_remediation_type(category: str, subject: str) -> RemediationType:
    cat = category.lower().strip()
    subj = subject.lower()
    if "mcp" in cat:
        return RemediationType.MCP_SWAP if "swap" in subj or "retire" in subj else RemediationType.NEW_INSTALL_MCP
    if "plugin" in cat:
        return RemediationType.PLUGIN_RETIRE if "retire" in subj or "drift" in subj else RemediationType.NEW_INSTALL_PLUGIN
    if "skill" in cat:
        return RemediationType.NEW_INSTALL_SKILL
    if "worktree" in cat or "worktree" in subj:
        return RemediationType.WORKTREE_PRUNE
    if "fqn" in subj:
        return RemediationType.FQN_REGEN
    if "cite" in subj or "stale-fact" in subj:
        return RemediationType.CITE_REFRESH
    if "settings" in cat or "settings" in subj:
        return RemediationType.SETTINGS_SURGERY
    return RemediationType.HIDDEN_ERROR_FIX

def parse_stream_findings(text: str, stream_id: str) -> List[Finding]:
    if not text.strip():
        return []
    if not _TABLE_HEADER_RE.search(text):
        raise StreamParseError(f"no findings table found in stream {stream_id}")
    findings: List[Finding] = []
    for m in _ROW_RE.finditer(text):
        ident, category, subject, evidence, risk, sca = (g.strip() for g in m.groups())
        sca_val = 0.0 if sca.lower() == "n/a" else float(sca)
        findings.append(Finding(
            id=ident,
            stream=stream_id.upper(),
            subject=subject,
            evidence=evidence,
            risk_class=_risk_to_class(risk),
            sca_score=sca_val,
            remediation_type=_infer_remediation_type(category, subject),
        ))
    return findings
```

```python
"""Inlined CoVe-factored verifier for W373 P2 synthesis.

Per Meta AI Dhuliawala+ arXiv:2309.11495, factored mode (each verification question answered
in isolated context, no draft access) outperforms joint/2-step by +8.4 / +4.2 pts on Wikidata
factoid hallucination.

This is the W373-ONLY stub. W374-EXT Task 7 lands the proper reusable agents/cove_verifier.py.

from __future__ import annotations
import re
from dataclasses import dataclass, field
from typing import List, Optional
from .finding_models import Finding

_QUESTION_GEN_PROMPT = """You are reviewing an audit finding for a SOTA-readiness wave.
Generate 3-5 sharp verification questions about this finding. Focus on:
- Is the evidence cite accurate + reachable?
- Is the risk-class proportional to the actual blast-radius?
- Could the finding be a false positive?
- Does the recommended remediation match the underlying issue?

Format your response as:
QUESTIONS:
1. <question>
2. <question>
...

Finding subject: {subject}
Finding evidence: {evidence}
Finding risk-class: {risk}
"""

_ANSWER_PROMPT = """Answer this verification question about a SOTA-readiness audit finding.
You do NOT have access to the finding's subject text — only the question.
If the question references specifics, answer based on the question alone.
Be candid: if the question reveals a flaw, say NO + explain.

Format: A<n>: <answer>

Question: {question}
"""

_RECONCILE_PROMPT = """You drafted a SOTA-readiness audit finding.
Below are verification questions you generated about it, and their independently-answered responses.
Identify discrepancies — places where the answers reveal flaws.

Format:
RECONCILED: <one of: "no discrepancies; finding accepted as-is" | "DISCREPANCY — <description>; needs re-eval">

Finding subject: {subject}
Finding evidence: {evidence}

Q/A pairs:
{qa_block}
"""

@dataclass
class CoVeResult:
    questions: List[str] = field(default_factory=list)
    answers: List[str] = field(default_factory=list)
    reconcile: str = ""
    flipped: bool = False  # True if reconcile contains DISCREPANCY

_Q_RE = re.compile(r"^\s*\d+\.\s*(.+?)\s*$", re.MULTILINE)
_A_RE = re.compile(r"^\s*A\d+:\s*(.+?)\s*$", re.MULTILINE)

async def cove_verify_finding(finding: Finding, anthropic_client, model: str = "claude-opus-4-7") -> CoVeResult:
    """Apply CoVe-factored verification to one Finding.

    Step 1: generate verification questions (draft VISIBLE in system prompt)
    Step 2: per Qi, get isolated answer (draft HIDDEN — factored mode)
    Step 3: reconcile (draft VISIBLE + Q/A pairs)
    """
    # Step 1: generate questions (draft visible)
    sys1 = _QUESTION_GEN_PROMPT.format(
        subject=finding.subject,
        evidence=finding.evidence,
        risk=finding.risk_class.name,
    )
    resp1 = await anthropic_client.messages.create(
        model=model,
        max_tokens=512,
        system=sys1,
        messages=[{"role": "user", "content": "Generate verification questions."}],
    )
    text1 = resp1.content[0].text
    questions = _Q_RE.findall(text1)

    # Step 2: per Qi, get isolated answer (draft HIDDEN — factored mode)
    answers: List[str] = []
    for q in questions:
        sys2 = _ANSWER_PROMPT.format(question=q)   # NO finding fields in system prompt
        resp2 = await anthropic_client.messages.create(
            model=model,
            max_tokens=256,
            system=sys2,
            messages=[{"role": "user", "content": "Answer the question."}],
        )
        answer_text = resp2.content[0].text
        match = _A_RE.search(answer_text)
        answers.append(match.group(1) if match else answer_text.strip())

    # Step 3: reconcile (draft visible + Q/A pairs)
    qa_block = "\n".join(f"Q{i+1}: {q}\nA{i+1}: {a}" for i, (q, a) in enumerate(zip(questions, answers)))
    sys3 = _RECONCILE_PROMPT.format(
        subject=finding.subject,
        evidence=finding.evidence,
        qa_block=qa_block,
    )
    resp3 = await anthropic_client.messages.create(
        model=model,
        max_tokens=512,
        system=sys3,
        messages=[{"role": "user", "content": "Reconcile."}],
    )
    reconcile_text = resp3.content[0].text.strip()

    flipped = "DISCREPANCY" in reconcile_text.upper()
    return CoVeResult(questions=questions, answers=answers, reconcile=reconcile_text, flipped=flipped)
```

```python
    return 1.5

    matches = _CITE_TOKEN_RE.findall(f.evidence)
    return min(5.0, 1.0 + 1.0 * len(matches))

def _d103_drift_detection_coverage(f: Finding) -> float:
    # Probe coverage proxy: does the evidence cite a re-runnable command?
    e = f.evidence.lower()
    score = 1.5
    for tok in ("git ", "npm view", "curl ", "gh api", "ls-remote", "test-netconnection", "get-service"):
        if tok in e:
            score += 0.5
    return min(5.0, score)

def _d104_vbc_compliance(f: Finding) -> float:
    # Higher if CoVe-verified + non-flipped
    if not f.cove_questions:
        return 1.0
    base = 3.0
    if f.cove_reconcile and "no discrepancies" in f.cove_reconcile.lower():
        base += 1.5
    if f.cove_flipped:
        base = max(0.5, base - 2.0)
    return min(5.0, base)

def _d105_pattern_density(f: Finding) -> float:
    # Goldilocks: subject length in 20-120 chars best
    n = len(f.subject)
    if 20 <= n <= 120:
        return 4.5
    if 10 <= n < 20 or 120 < n <= 200:
        return 3.0
    return 1.5

def score_finding(f: Finding) -> ScaV18Result:
    components = {
        D101_NAME: _d101_inverse_dep_density(f),
        D102_NAME: _d102_citation_chain_depth(f),
        D103_NAME: _d103_drift_detection_coverage(f),
        D104_NAME: _d104_vbc_compliance(f),
        D105_NAME: _d105_pattern_density(f),
    }
    aggregate = sum(components.values()) / 5.0
    return ScaV18Result(aggregate=round(aggregate, 2), components=components)
```

```python
"""Topo-sort + Markdown manifest renderer for W373 P2 output."""
from __future__ import annotations
from graphlib import TopologicalSorter
from typing import Iterable, List
from .finding_models import Finding, RiskClass

def topo_sort_findings(findings: Iterable[Finding]) -> List[Finding]:
    """Topologically sort by Finding.depends_on.

    Raises CycleError if a cycle exists.
    """
    by_id = {f.id: f for f in findings}
    ts = TopologicalSorter()
    for f in by_id.values():
        ts.add(f.id, *[d for d in f.depends_on if d in by_id])
    ordered_ids = list(ts.static_order())
    # Stable within topo level: preserve insertion order
    return [by_id[i] for i in ordered_ids if i in by_id]

def render_manifest_md(findings: List[Finding]) -> str:
    lines = [
        "# W373 Remediation Manifest",
        "",
        "Generated by `tools/w373/P2-synthesis.py` (Phase 2 output).",
        "Order is topological; each row's `depends_on` precedes it.",
        "",
        "## Remediation Manifest",
        "",
        "| # | ID | Stream | Risk | sca | Remediation type | Subject | Depends on |",
        "|---|------|--------|------|-----|------------------|---------|------------|",
    ]
    for idx, f in enumerate(findings, start=1):
        deps = ", ".join(f.depends_on) if f.depends_on else "-"
        lines.append(
            f"| {idx} | {f.id} | {f.stream} | {f.risk_class.name} | "
            f"{f.sca_score:.2f} | {f.remediation_type.value} | "
            f"{f.subject} | {deps} |"
        )
    # Per-class summary
    n_low = sum(1 for f in findings if f.risk_class is RiskClass.LOW)
    n_high = sum(1 for f in findings if f.risk_class is RiskClass.HIGH)
    lines.append("")
    lines.append(f"**Summary:** {n_low} LOW (auto-fire) · {n_high} HIGH (jury-gated)")
    lines.append("")
    return "\n".join(lines)
```

```python
"""W373 Phase 2 — Synthesis with CoVe-factored verification + sca-v18 + topo-sort.

Reads 6 stream output files under <workdir>/.claude/state/W373-stream-{A..F}-*.md,
runs CoVe-factored verification on each finding (unless --skip-cove), scores via
sca-v18 D101-D105, classifies risk LOW/HIGH per Q2 settled rule, topo-sorts by
depends_on, and writes:
  - <workdir>/docs/architecture/W373-SOTA-READY/W373-FINDINGS.md
  - <workdir>/docs/architecture/W373-SOTA-READY/REMEDIATION-MANIFEST.md
"""
from __future__ import annotations
import argparse
import asyncio
import json
import os
import pathlib
import sys
from typing import List

# Allow tools.w373.lib imports when run from worktree root
sys.path.insert(0, str(pathlib.Path(__file__).resolve().parents[2]))

from tools.w373.lib.finding_models import Finding, RiskClass
from tools.w373.lib.stream_parser import parse_stream_findings
from tools.w373.lib.cove_verifier import cove_verify_finding
from tools.w373.lib.sca_v18_scorer import score_finding
from tools.w373.lib.risk_classifier import classify_risk_final
from tools.w373.lib.manifest_writer import topo_sort_findings, render_manifest_md

def _load_streams(state_dir: pathlib.Path) -> List[Finding]:
    findings: List[Finding] = []
    for stream_id in "ABCDEF":
        pattern = f"W373-stream-{stream_id}-*.md"
        matches = list(state_dir.glob(pattern))
        if not matches:
            print(f"[P2] WARN: no stream {stream_id} output found at {state_dir}/{pattern}; coverage gap noted")
            continue
        text = matches[0].read_text(encoding="utf-8")
        try:
            findings.extend(parse_stream_findings(text, stream_id=stream_id))
        except Exception as e:
            print(f"[P2] WARN: stream {stream_id} parse error: {e}; skipping")
    return findings

async def _maybe_cove(findings: List[Finding], skip: bool) -> None:
    if skip:
        return
    try:
        import anthropic
    except ImportError:
        print("[P2] anthropic not installed; --skip-cove implicit")
        return
    api_key = os.environ.get("ANTHROPIC_API_KEY")
    if not api_key:
        print("[P2] ANTHROPIC_API_KEY not set; --skip-cove implicit")
        return
    client = anthropic.AsyncAnthropic(api_key=api_key)
    for f in findings:
        try:
            res = await cove_verify_finding(f, client)
            f.cove_questions = res.questions
            f.cove_answers = res.answers
            f.cove_reconcile = res.reconcile
            f.cove_flipped = res.flipped
        except Exception as e:
            print(f"[P2] CoVe error on {f.id}: {e}; finding stays unverified")

def _render_findings_md(findings: List[Finding]) -> str:
    lines = [
        "# W373-FINDINGS.md",
        "",
        f"Total findings: {len(findings)}",
        "",
        "## All findings (pre-remediation)",
        "",
        "| ID | Stream | Subject | Evidence | Risk | sca | CoVe-flipped | Remediation type |",
        "|----|--------|---------|----------|------|-----|--------------|------------------|",
    ]
    for f in findings:
        cove_flag = "YES" if f.cove_flipped else ("VERIFIED" if f.cove_questions else "skipped")
        lines.append(
            f"| {f.id} | {f.stream} | {f.subject} | {f.evidence} | "
            f"{f.risk_class.name} | {f.sca_score:.2f} | {cove_flag} | "
            f"{f.remediation_type.value} |"
        )
    return "\n".join(lines)

def main():
    p = argparse.ArgumentParser()
    p.add_argument("--workdir", default="Z:/claude-sota-installed-W373",
                   help="Worktree root (default: %(default)s)")
    p.add_argument("--skip-cove", action="store_true",
                   help="Skip CoVe-factored verification (faster; for tests/fixtures)")
    args = p.parse_args()

    workdir = pathlib.Path(args.workdir)
    state_dir = workdir / ".claude" / "state"
    out_dir = workdir / "docs" / "architecture" / "W373-SOTA-READY"
    out_dir.mkdir(parents=True, exist_ok=True)

    print(f"[P2] Loading streams from {state_dir}")
    findings = _load_streams(state_dir)
    print(f"[P2] Parsed {len(findings)} findings across all streams")

    # CoVe verification (skipped for fixtures/tests)
    asyncio.run(_maybe_cove(findings, skip=args.skip_cove))

    # sca-v18 scoring + risk re-class
    for f in findings:
        sca = score_finding(f)
        f.sca_score = sca.aggregate
        f.sca_v18_components = sca.components
        f.risk_class = classify_risk_final(f, sca)

    # Topo-sort
    ordered = topo_sort_findings(findings)

    findings_md = _render_findings_md(ordered)
    manifest_md = render_manifest_md(ordered)

    (out_dir / "W373-FINDINGS.md").write_text(findings_md, encoding="utf-8")
    (out_dir / "REMEDIATION-MANIFEST.md").write_text(manifest_md, encoding="utf-8")

    n_low = sum(1 for f in ordered if f.risk_class is RiskClass.LOW)
    n_high = sum(1 for f in ordered if f.risk_class is RiskClass.HIGH)
    print(f"[P2] WROTE: {out_dir}/W373-FINDINGS.md ({len(ordered)} findings)")
    print(f"[P2] WROTE: {out_dir}/REMEDIATION-MANIFEST.md ({n_low} LOW + {n_high} HIGH)")
    return 0

if __name__ == "__main__":
    sys.exit(main())
```

```python
"""codex CLI subprocess wrapper — Path P (foreground+tee).

Cite: W372 P0-2 codex r1+r2 pattern; .claude/plugins/cache/openai-codex/codex/1.0.4/scripts/codex-companion.mjs.
"""
from __future__ import annotations
import json
import os
import re
import subprocess
from dataclasses import dataclass
from pathlib import Path
from typing import Optional

class CodexUnavailable(RuntimeError):
    pass

@dataclass
class CodexVerdict:
    kind: str           # "APPROVE" | "NEEDS_REVISION" | "BLOCK" | "REVISE"
    confidence: float   # 0.0 - 1.0
    rationale: str
    raw: str

_VERDICT_RE = re.compile(r"^\s*Verdict:\s*(APPROVE|NEEDS_REVISION|BLOCK|REVISE|SHIP)\s*$", re.MULTILINE)
_CONFIDENCE_RE = re.compile(r"^\s*Confidence:\s*([\d.]+)\s*$", re.MULTILINE)

_DEFAULT_COMPANION = Path(
    "Z:/claude-sota-installed/.claude/plugins/cache/openai-codex/codex/1.0.4/scripts/codex-companion.mjs"
)

class CodexClient:
    def __init__(self, companion_path: Optional[Path] = None, node_bin: str = "Z:/tools/nodejs/node.exe"):
        self.companion_path = companion_path or _DEFAULT_COMPANION
        self.node_bin = node_bin

    def task_review(self, prompt: str, effort: str = "high", timeout: int = 900) -> CodexVerdict:
        """Dispatch a codex review via the companion script (Path P foreground+tee)."""
        cmd = [self.node_bin, str(self.companion_path), "task", "--effort", effort, prompt]
        try:
            result = subprocess.run(cmd, capture_output=True, text=True, timeout=timeout)
        except FileNotFoundError as e:
            raise CodexUnavailable(f"codex companion script not found at {self.companion_path}: {e}")
        if result.returncode != 0:
            raise CodexUnavailable(f"codex returned exit {result.returncode}: {result.stderr[:500]}")
        stdout = result.stdout
        verdict_m = _VERDICT_RE.search(stdout)
        conf_m = _CONFIDENCE_RE.search(stdout)
        kind = verdict_m.group(1) if verdict_m else "BLOCK"
        confidence = float(conf_m.group(1)) if conf_m else 0.0
        # extract rationale (best-effort)
        rationale_lines = []
        for line in stdout.splitlines():
            if line.strip().lower().startswith("rationale"):
                rationale_lines.append(line.split(":", 1)[1].strip() if ":" in line else "")
        rationale = " ".join(rationale_lines) if rationale_lines else stdout[-500:]
        return CodexVerdict(kind=kind, confidence=confidence, rationale=rationale, raw=stdout)
```

```python
"""Inlined 3-judge codex jury for W373 HIGH-risk remediation gating.

Per Zheng+ arXiv:2306.05685 (MT-Bench position-swap) + Wang+ arXiv:2310.17631 (JudgeLM
CISC confidence-weighted aggregation) + OpenReview 2025 id XdcofpTCyq.

W373-only stub. W374-EXT Task 9 lands the proper reusable agents/jury_aggregator.py +
tools/codex-jury-panel.mjs.
"""
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Dict, List
from .finding_models import JuryVerdict, PanelResult, JuryDecision
from .codex_client import CodexClient, CodexVerdict

@dataclass
class PanelInput:
    diff: str                      # the staged remediation diff
    finding_id: str
    cite_anchors: Dict[str, str]   # {"A": "<stream A anchor>", "B": "<...>", "C": "<...>"}

_PANEL_PROMPT_TEMPLATE = """[Position-swap prefix: {cite_anchor}]

You are panel judge {panel_id} (temperature={temperature}). Review the following remediation
diff for finding {finding_id}. Verdict: APPROVE, REVISE, BLOCK, or NEEDS_ROUND_2.

Diff:
```
{diff}
```

Format your output exactly:
Verdict: <APPROVE|REVISE|BLOCK|NEEDS_ROUND_2>
Confidence: <float 0.0-1.0>
Rationale: <one paragraph>
"""

# Maps codex verdict strings → JuryVerdict
_VERDICT_MAP = {
    "APPROVE": JuryVerdict.APPROVE,
    "REVISE": JuryVerdict.REVISE,
    "BLOCK": JuryVerdict.BLOCK,
    "NEEDS_ROUND_2": JuryVerdict.NEEDS_ROUND_2,
    "NEEDS_REVISION": JuryVerdict.REVISE,
    "SHIP": JuryVerdict.APPROVE,
}

def _codex_verdict_to_jury(cv: CodexVerdict) -> JuryVerdict:
    return _VERDICT_MAP.get(cv.kind, JuryVerdict.BLOCK)

def dispatch_jury_panel(pin: PanelInput, codex_client: CodexClient) -> JuryDecision:
    """Dispatch 3 codex rounds at temps 0.0/0.3/0.7 with position-swap."""
    configs = [
        ("alpha", 0.0, "A"),
        ("beta",  0.3, "B"),
        ("gamma", 0.7, "C"),
    ]
    panels: List[PanelResult] = []
    for panel_id, temp, swap in configs:
        prompt = _PANEL_PROMPT_TEMPLATE.format(
            cite_anchor=pin.cite_anchors.get(swap, ""),
            panel_id=panel_id,
            temperature=temp,
            finding_id=pin.finding_id,
            diff=pin.diff,
        )
        cv = codex_client.task_review(prompt, effort="high")
        panels.append(PanelResult(
            panel_id=panel_id,
            verdict=_codex_verdict_to_jury(cv),
            confidence=cv.confidence,
            rationale=cv.rationale,
            position_swap=swap,
        ))
    final = aggregate_cisc(panels)
    return JuryDecision(finding_id=pin.finding_id, final_verdict=final, panels=panels, aggregation="CISC-weighted-majority")

def aggregate_cisc(panels: List[PanelResult]) -> JuryVerdict:
    """CISC: confidence-weighted majority per Wang+ arXiv:2310.17631.

    If any verdict's confidence sum > 1.5 (≥ 2 strong votes worth), it wins.
    If max confidence sum < 1.5 AND all panels disagree, NEEDS_ROUND_2.
    """
    by_verdict: Dict[JuryVerdict, float] = {}
    for p in panels:
        by_verdict[p.verdict] = by_verdict.get(p.verdict, 0.0) + p.confidence
    if not by_verdict:
        return JuryVerdict.BLOCK
    top_verdict, top_conf = max(by_verdict.items(), key=lambda kv: kv[1])
    if top_conf >= 1.5:
        return top_verdict
    # weak split → need round 2
    return JuryVerdict.NEEDS_ROUND_2
```

```python
class ManifestRow:
    __slots__ = ("id", "stream", "risk", "sca", "rtype", "subject")
    def __init__(self, id, stream, risk, sca, rtype, subject):
        self.id, self.stream = id, stream

        self.sca = float(sca)

    text = path.read_text(encoding="utf-8")
    rows = []

    return rows

def _execute_low(row: ManifestRow, dry_run: bool) -> dict:

    if dry_run:

    return f"[stub diff for HIGH finding {row.id} — actual diff produced by remediation handler]"

    manifest_path = pathlib.Path(args.manifest)

        manifest_path = workdir / args.manifest
    log_out = pathlib.Path(args.log_out)
    if not log_out.is_absolute():
        log_out = workdir / args.log_out
    log_out.parent.mkdir(parents=True, exist_ok=True)
    jury_out = pathlib.Path(args.jury_out)
    if not jury_out.is_absolute():
        jury_out = workdir / args.jury_out
    jury_out.parent.mkdir(parents=True, exist_ok=True)

    rows = parse_manifest(manifest_path)
    print(f"[P3] Loaded {len(rows)} findings from {manifest_path}")
    if not rows:
        log_out.write_text("# W373 Remediation Log\n\nNo findings found. Manifest empty.\n", encoding="utf-8")
        return 0

    cc: Optional[CodexClient] = None
    if not args.dry_run:
        try:
            cc = CodexClient()
        except CodexUnavailable as e:
            print(f"[P3] codex unavailable, HIGH findings will SKIP jury: {e}")

    log_lines = [
        "# W373 Remediation Log",
        f"\nGenerated: {datetime.now(timezone.utc).isoformat()}",
        f"Mode: {'dry-run' if args.dry_run else 'LIVE'}",
        "",
        "| # | ID | Risk | Action | Outcome | Detail |",
        "|---|------|------|--------|---------|--------|",
    ]
    jury_log = []

    for idx, row in enumerate(rows, start=1):
        if row.risk is RiskClass.LOW:
            r = _execute_low(row, dry_run=args.dry_run)
            outcome = "OK" if r["ok"] else "MANUAL"
            log_lines.append(f"| {idx} | {row.id} | LOW | {r['action']} | {outcome} | {r.get('reason', '-')} |")
            continue
        # HIGH path
        diff = _stage_diff_for_high(row)
        if args.dry_run or cc is None:
            log_lines.append(f"| {idx} | {row.id} | HIGH | DRY_RUN_SKIP_JURY | skipped | - |")
            continue
        cite_anchors = {"A": f"stream-A:{row.id}", "B": f"stream-B:{row.id}", "C": f"stream-C:{row.id}"}
        pin = PanelInput(diff=diff, finding_id=row.id, cite_anchors=cite_anchors)
        try:
            decision = dispatch_jury_panel(pin, codex_client=cc)
            jury_log.append({
                "finding_id": row.id,
                "final_verdict": decision.final_verdict.value,
                "panels": [
                    {"id": pp.panel_id, "verdict": pp.verdict.value, "confidence": pp.confidence}
                    for pp in decision.panels
                ],
            })
            log_lines.append(f"| {idx} | {row.id} | HIGH | JURY_{decision.final_verdict.value} | "
                             f"{decision.final_verdict.value} | 3 panels |")
        except Exception as e:
            log_lines.append(f"| {idx} | {row.id} | HIGH | JURY_ERROR | error | {e} |")

    log_out.write_text("\n".join(log_lines) + "\n", encoding="utf-8")
    if jury_log:
        jury_out.write_text("\n".join(json.dumps(j) for j in jury_log) + "\n", encoding="utf-8")
    print(f"[P3] WROTE: {log_out}")
    if jury_log:
        print(f"[P3] WROTE: {jury_out} ({len(jury_log)} jury decisions)")
    return 0

if __name__ == "__main__":
    sys.exit(main())
```
